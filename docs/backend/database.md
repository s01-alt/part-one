# Database schema — DRAFT

> **Status: Draft** — derived from the PRD and the frontend types in
> `src/lib/types.ts`. PostgreSQL syntax. Final DDL will be generated by the
> ORM's migrations once the stack is approved (D-040).

## Global rules

- **IDs**: `text` primary keys with type prefixes generated by the app
  (`usr_`, `brd_`, `prd_`, `var_`, `ord_`, `so_`, `pay_`, `led_` …) using a
  sortable random ID (e.g. ULID). Never expose sequential integers.
- **Money**: `bigint` kobo, `CHECK (amount >= 0)` unless the column is a signed ledger amount.
- **Rates**: `numeric(5,4)` (0.1000 = 10%).
- **Timestamps**: `timestamptz`, `created_at default now()`, `updated_at` maintained by the app.
- **Soft delete** only where history matters (products, comments, reviews): `deleted_at timestamptz`.
- **Snapshots**: orders copy product name, variant label, price, commission rate and
  brand name at purchase time. Later edits never change past orders.
- **Human order numbers**: `marketplace_orders.number` from a Postgres sequence
  (1001, 1002 …); seller orders append a letter (`1001-A`).

## Entity map

```
users ─┬─< sessions / oauth_accounts
       ├─< carts ─< cart_items >─ product_variants
       ├─< marketplace_orders ─< seller_orders ─< order_lines
       │                               └─< order_status_events
       ├─< likes / saves / reposts / shares / comments >─ products
       ├─< brand_follows >─ brands
       ├─< reviews >─ products (requires delivered order_line)
       ├─< reports
       └─< notifications

brands ─┬─< products ─┬─< product_variants
        │             ├─< product_images
        │             └─<> collections (collection_products)
        ├─< seller_orders
        ├─< ledger_entries
        ├─< payouts
        └── bank_accounts (1 active)

payments (1 per marketplace_order attempt) ─< refunds, chargebacks
webhook_events · audit_log · platform_settings · analytics_events
```

## Tables

### Identity and access

```sql
create table users (
  id               text primary key,
  email            citext not null unique,
  name             text not null,
  avatar_url       text,
  seller_brand_id  text unique references brands(id),   -- one login per brand (D-030)
  admin_role       text check (admin_role in ('founder','report_manager','developer')),
  status           text not null default 'active' check (status in ('active','suspended')),
  created_at       timestamptz not null default now(),
  updated_at       timestamptz not null default now()
);

create table oauth_accounts (
  provider          text not null,              -- 'google'
  provider_user_id  text not null,              -- Google 'sub'
  user_id           text not null references users(id),
  primary key (provider, provider_user_id)
);

create table sessions (
  id            text primary key,               -- random 256-bit, stored hashed
  token_hash    bytea not null unique,
  user_id       text not null references users(id),
  expires_at    timestamptz not null,
  last_seen_at  timestamptz not null default now(),
  user_agent    text,
  ip            inet,
  revoked_at    timestamptz
);

create table addresses (
  id text primary key, user_id text not null references users(id),
  full_name text not null, phone text not null, line1 text not null, line2 text,
  city text not null, state text not null, is_default boolean not null default false
);
```

### Catalogue

```sql
create table products (
  id text primary key,
  brand_id text not null references brands(id),
  slug citext not null unique,
  name text not null, description text not null,
  type text not null,                    -- t-shirt, hoodie … 'other'
  styles text[] not null default '{}',   -- baggy, minimalist, y2k …
  material text, care_instructions text,
  price bigint not null check (price >= 0),          -- VAT exclusive
  status text not null default 'active' check (status in ('active','inactive','removed')),
  removed_reason text,                                -- set by admin moderation
  rating_avg numeric(3,2) not null default 0, rating_count int not null default 0,
  like_count int not null default 0, save_count int not null default 0,
  repost_count int not null default 0, share_count int not null default 0,
  comment_count int not null default 0,
  total_stock int not null default 0,                 -- maintained from variants
  search tsvector generated always as (
    setweight(to_tsvector('simple', coalesce(name,'')), 'A') ||
    setweight(to_tsvector('simple', array_to_string(styles,' ') || ' ' || type), 'B') ||
    setweight(to_tsvector('simple', coalesce(description,'')), 'C')
  ) stored,
  created_at timestamptz not null default now(), updated_at timestamptz not null default now(),
  deleted_at timestamptz
);
create index products_search on products using gin (search);
create index products_name_trgm on products using gin (name gin_trgm_ops);
create index products_listing on products (status, created_at desc) where deleted_at is null;
create index products_brand on products (brand_id);

create table product_variants (
  id text primary key, product_id text not null references products(id),
  sku text not null, size text not null,
  color_name text not null, color_hex text not null,
  price bigint check (price >= 0),                 -- null → product price
  stock int not null default 0 check (stock >= 0), -- never negative (D-015)
  unique (product_id, size, color_name)
);

create table product_images (
  id text primary key, product_id text not null references products(id),
  storage_key text not null, alt text, position smallint not null check (position between 0 and 4)
);   -- max 5 images enforced by position + unique below
create unique index product_image_position on product_images(product_id, position);

create table collections (
  id text primary key, brand_id text not null references brands(id),
  slug citext not null, name text not null, description text,
  unique (brand_id, slug)
);
create table collection_products (
  collection_id text references collections(id), product_id text references products(id),
  position int not null default 0, primary key (collection_id, product_id)
);
```

**Listed** (visible in the marketplace) = `products.status = 'active'` AND
`deleted_at is null` AND `total_stock > 0` AND brand `status = 'approved'`.

create sequence order_number_seq start 1001;

create table marketplace_orders (

create table seller_orders (
  id text primary key,
  marketplace_order_id text not null references marketplace_orders(id),
  brand_id text not null references brands(id),

    check (status in ('pending_payment','paid','processing','shipped','delivered',
                      'cancelled','returned','refunded')),
  items_subtotal bigint not null, shipping_fee bigint not null, vat bigint not null, total bigint not null,
  commission_rate numeric(5,4) not null,           -- snapshot (D-022)
  commission bigint not null,
  seller_earnings bigint not null,                 -- items − commission + shipping
  brand_name text not null,                        -- snapshot
  delivered_at timestamptz,
  unique (marketplace_order_id, brand_id),
  unique (marketplace_order_id, suffix)
);

create table order_lines (
  id text primary key, seller_order_id text not null references seller_orders(id),
  product_id text not null references products(id), variant_id text not null references product_variants(id),
  product_name text not null, variant_label text not null,       -- snapshots

  id text primary key, seller_order_id text not null references seller_orders(id),
  from_status text, to_status text not null,
  actor_type text not null check (actor_type in ('customer','seller','admin','system')),
  actor_id text, note text, created_at timestamptz not null default now()

### Payments, refunds, chargebacks, webhooks

```sql
create table payments (
  id text primary key, marketplace_order_id text not null references marketplace_orders(id),
  provider text not null default 'paystack',
  reference text not null unique,                   -- our reference sent to Paystack
  amount bigint not null, currency char(3) not null default 'NGN',
  status text not null check (status in ('initialized','success','failed','abandoned')),
  provider_transaction_id text, channel text, card_last4 text,

create table refunds (
  id text primary key, payment_id text not null references payments(id),
  seller_order_id text references seller_orders(id),

  reason text not null check (reason in ('customer_cancel','return','oversell','admin','chargeback')),
  status text not null check (status in ('pending','processing','processed','failed')),

create table chargebacks (
  id text primary key, payment_id text not null references payments(id),
  amount bigint not null, status text not null, provider_dispute_id text unique,
  raw jsonb, created_at timestamptz not null default now()
);

create table webhook_events (
  id text primary key,
  provider text not null, event_type text not null,
  provider_event_key text not null,            -- e.g. event + data.id/reference
  payload jsonb not null, signature_valid boolean not null,
  status text not null default 'received' check (status in ('received','processed','ignored','failed')),
  error text, received_at timestamptz not null default now(), processed_at timestamptz,
  unique (provider, provider_event_key)        -- idempotency

```

### Seller ledger and payouts

The ledger is **append-only**. Balances are sums, never stored counters that can drift.

```sql
create table ledger_entries (
  id text primary key, brand_id text not null references brands(id),
  type text not null check (type in ('sale','commission','shipping','refund','chargeback','payout','payout_reversal','adjustment')),
  amount bigint not null,                       -- signed
  seller_order_id text references seller_orders(id),
  payout_id text references payouts(id),
  refund_id text references refunds(id),
  available_at timestamptz,                     -- null = pending; set when delivered (+ hold, D-026)
  description text not null,
  created_by text references users(id),         -- for admin adjustments
  created_at timestamptz not null default now()
);
create index on ledger_entries (brand_id, created_at desc);

create table payouts (
  id text primary key, brand_id text not null references brands(id),
  bank_account_id text not null references bank_accounts(id),
  amount bigint not null check (amount > 0),
  status text not null check (status in ('queued','processing','paid','failed','reversed')),
  transfer_reference text not null unique, provider_transfer_code text,
  failure_reason text, created_at timestamptz not null default now(), completed_at timestamptz
);
-- at most one in-flight payout per brand
create unique index one_open_payout on payouts(brand_id) where status in ('queued','processing');
```

- **Available balance** = `sum(amount) where available_at <= now()` (payout entries are negative and immediately "available").
- **Pending balance** = `sum(amount) where available_at is null`.
- A payout may only be created inside a transaction that locks the brand row and checks available ≥ `platform_settings.payout_threshold`.

### Social, reviews, moderation, notifications

```sql
create table likes  (user_id text references users(id), product_id text references products(id),
                     created_at timestamptz not null default now(), primary key (user_id, product_id));
create table saves  (user_id text references users(id), product_id text references products(id),
                     created_at timestamptz not null default now(), primary key (user_id, product_id));
create table reposts (id text primary key, user_id text not null references users(id),
                      product_id text not null references products(id), caption text check (length(caption) <= 280),
                      created_at timestamptz not null default now(), deleted_at timestamptz,
                      unique (user_id, product_id));
create index on reposts (created_at desc) where deleted_at is null;   -- global feed
create table shares (id text primary key, user_id text references users(id),   -- nullable: guests can share
                     product_id text not null references products(id), channel text,
                     created_at timestamptz not null default now());
create table brand_follows (user_id text references users(id), brand_id text references brands(id),
                            created_at timestamptz not null default now(), primary key (user_id, brand_id));

create table comments (
  id text primary key, product_id text not null references products(id),
  user_id text not null references users(id),
  parent_id text references comments(id),          -- one level of replies (optional)
  body text not null check (length(body) between 1 and 1000),
  status text not null default 'visible' check (status in ('visible','removed')),
  created_at timestamptz not null default now(), deleted_at timestamptz
);

create table reviews (
  id text primary key, product_id text not null references products(id),
  user_id text not null references users(id),
  order_line_id text not null unique references order_lines(id),  -- proof of purchase
  rating smallint not null check (rating between 1 and 5),
  body text check (length(body) <= 2000),
  status text not null default 'visible' check (status in ('visible','removed')),
  created_at timestamptz not null default now()
);
-- App rule: allowed only when the line's seller_order.status = 'delivered' (D-032).

create table reports (
  id text primary key, reporter_id text not null references users(id),
  target_type text not null check (target_type in ('product','review','comment')),
  target_id text not null,
  reason text not null check (reason in ('counterfeit','offensive','spam','misleading','other')),
  details text, status text not null default 'open' check (status in ('open','actioned','dismissed')),
  handled_by text references users(id), handled_at timestamptz,
  created_at timestamptz not null default now(),
  unique (reporter_id, target_type, target_id)
);

create table notifications (
  id text primary key, user_id text not null references users(id),
  type text not null, title text not null, body text not null, href text,
  read_at timestamptz, created_at timestamptz not null default now()
);
create index on notifications (user_id, created_at desc);
```

Counters (`like_count`, …) are updated in the same transaction as the insert/delete.
A nightly job recomputes them to fix any drift.

### Platform, audit, analytics

```sql
create table platform_settings (
  id boolean primary key default true check (id),  -- single row
  default_commission_rate numeric(5,4) not null,
  payout_threshold bigint not null default 5000000,  -- ₦50,000 (D-024)
  vat_rate numeric(5,4) not null default 0.0750,
  payout_hold_days int not null default 0,           -- D-026
  updated_by text references users(id), updated_at timestamptz not null default now()
);

create table audit_log (
  id text primary key, actor_id text references users(id), actor_role text,
  action text not null,                 -- e.g. 'refund.create', 'seller.suspend'
  target_type text not null, target_id text not null,
  before jsonb, after jsonb, ip inet, created_at timestamptz not null default now()
);   -- append-only: app DB role has INSERT/SELECT only

create table analytics_events (   -- outbox → PostHog, and source for recommendations
  id text primary key, user_id text, anonymous_id text,
  event text not null,             -- product_viewed, product_liked, checkout_started …
  product_id text, brand_id text, properties jsonb not null default '{}',
  created_at timestamptz not null default now(), exported_at timestamptz
);
create index on analytics_events (created_at) where exported_at is null;
```
